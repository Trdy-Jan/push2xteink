# P2a 抓取（feeds + extract）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** 实现 RSS 拉取 + 去重（`feeds.py`）和文章正文提取 + 回退（`extract.py`），产出 `Article` 列表供后续流水线使用。

**Architecture:** `feeds.py` 用 httpx 抓 RSS 字节流、feedparser 解析成 `Article`，再用 P1 的 `State` 去重（含窗口内重试）。`extract.py` 用 httpx 抓文章页、trafilatura 提取正文，失败/过短则返回 `None` 让调用方回退到 RSS 内容。两模块互不依赖，均可独立测试；HTTP 全部用 `respx` mock，RSS/HTML 用本地 fixture。

**Tech Stack:** Python 3.12、httpx、feedparser、trafilatura、pytest、respx。

**Spec:** `docs/superpowers/specs/2026-08-31-push2xteink-design.md`（第 3、6 节步骤 1–3；模块表第 62–63 行）

## Global Constraints

- Python `>=3.12`；类型标注 `X | None` / `list[...]`，不用 `Optional` / `List`。
- 所有 HTTP 请求用 `httpx`，显式传 `timeout=`；代理用 `httpx.Client(proxy=proxy_url)`（httpx `>=0.27` 的参数名是 `proxy`，不是 `proxies`）。
- `Article` 来自 `push2xteink.models`（P1 已实现），**不可修改**。字段：`feed_id, guid, title, link, published_at: datetime|None, author, source_title, content_html, content_is_full_text: bool, summary`。
- `Feed` / `State` 来自 P1。`State.is_item_pushable(feed_id, guid, lookback_hours, *, now=None) -> bool`、`State.record_seen(feed_id, guid, *, now=None)`。
- 所有 datetime 为 tz-aware UTC。
- 测试不发真实网络请求——httpx 用 `respx` mock，RSS/HTML 用 `tests/fixtures/` 下的本地文件。
- 源文件 `src/push2xteink/`，测试 `tests/`。

```python
from __future__ import annotations

import calendar
import dataclasses
from datetime import datetime, timezone

import feedparser
import httpx

from .models import Article, Feed

_UA = {"User-Agent": "push2xteink/0.1 (+https://github.com/)"}


@dataclasses.dataclass
class FeedResult:
    articles: list[Article] = dataclasses.field(default_factory=list)
    error: str | None = None


def _struct_to_utc(st) -> datetime | None:
    if not st:
        return None
    return datetime.fromtimestamp(calendar.timegm(st), tz=timezone.utc)


def _guid(entry) -> str:
    return entry.get("id") or entry.get("guid") or entry.get("link") or ""


def _content_html(entry) -> str:
    contents = entry.get("content")
    if contents:
        return contents[0].get("value", "") or ""
    return entry.get("summary", "") or ""


def fetch_feed(
    feed: Feed, *, proxy_url: str | None = None, timeout: float = 20.0
) -> FeedResult:
    try:
        with httpx.Client(
            proxy=proxy_url, timeout=timeout, follow_redirects=True, headers=_UA
        ) as client:
            resp = client.get(feed.url)
            resp.raise_for_status()
        raw = resp.content
    except httpx.HTTPError as exc:
        return FeedResult(error=f"fetch failed: {exc!s}")

    parsed = feedparser.parse(raw)
    if parsed.bozo and not parsed.entries:
        return FeedResult(error=f"parse failed: {parsed.get('bozo_exception')!s}")

    source_title = parsed.feed.get("title")
    articles: list[Article] = []
    for entry in parsed.entries:
        guid = _guid(entry)
        if not guid:
            continue
        articles.append(
            Article(
                feed_id=feed.id,
                guid=guid,
                title=entry.get("title") or "(untitled)",
                link=entry.get("link") or "",
                published_at=_struct_to_utc(
                    entry.get("published_parsed") or entry.get("updated_parsed")
                ),
                author=entry.get("author"),
                source_title=source_title,
                content_html=_content_html(entry),
                content_is_full_text=False,
            )
        )
    return FeedResult(articles=articles)


def select_new_articles(*args, **kwargs):  # Task 2
    raise NotImplementedError
```

```python
from datetime import timedelta  # 加到 import


def select_new_articles(
    state: "State",
    feed_id: str,
    articles: list[Article],
    *,
    first_run: bool,
    lookback_hours: int,
    now: datetime | None = None,
) -> list[Article]:
    now = now or datetime.now(timezone.utc)
    if now.tzinfo is None:
        now = now.replace(tzinfo=timezone.utc)
    cutoff = now - timedelta(hours=lookback_hours)

    kept: list[Article] = []
    for art in articles:
        if not state.is_item_pushable(feed_id, art.guid, lookback_hours, now=now):
            continue
        if first_run and (art.published_at is None or art.published_at < cutoff):
            continue
        state.record_seen(feed_id, art.guid, now=now)
        kept.append(art)
    return kept
```

```python
from __future__ import annotations

from html import escape

import httpx
import trafilatura

from .models import Article

_UA = {"User-Agent": "push2xteink/0.1 (+https://github.com/)"}


def extract_full_text(
    url: str,
    *,
    proxy_url: str | None = None,
    timeout: float = 20.0,
    min_chars: int = 200,
) -> str | None:
    if not url:
        return None
    try:
        with httpx.Client(
            proxy=proxy_url, timeout=timeout, follow_redirects=True, headers=_UA
        ) as client:
            resp = client.get(url)
            resp.raise_for_status()
        html = resp.text
    except httpx.HTTPError:
        return None

    text = trafilatura.extract(
        html, include_comments=False, include_tables=False, url=url or None
    )
    if not text or len(text) < min_chars:
        return None
    paras = [escape(p.strip()) for p in text.split("\n") if p.strip()]
    return "".join(f"<p>{p}</p>" for p in paras)

    raise NotImplementedError
```

```python
def apply_full_text(
    article: Article,
    *,
    enabled: bool,
    proxy_url: str | None = None,
    timeout: float = 20.0,
) -> Article:
    if not enabled:
        return article
    extracted = extract_full_text(article.link, proxy_url=proxy_url, timeout=timeout)
    if extracted is None:
        return article
    return article.model_copy(
        update={"content_html": extracted, "content_is_full_text": True}
    )
```
