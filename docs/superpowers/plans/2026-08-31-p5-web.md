# P5 Web 管理界面 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** 实现 FastAPI + HTMX 的轻量 Web 管理界面——增删改 RSS 源 / 推送任务、看执行历史、手动触发、编辑连接设置、测试连接——并把 `serve` 入口改成 uvicorn 托管 FastAPI + 调度器同进程（spec §3/§10）。

**Architecture:** `web/app.py` 建一个 `FastAPI` app：lifespan 里起 `State` + `Scheduler` + `_ConfigWatcher`（守护线程，5s 轮询 `config.yaml` mtime 触发 `scheduler.maybe_reload`），关停时反序清理。所有配置改动走「载入当前 `Config` → 应用改动 → 整体重校验 → `config.py` 的 `write_config` 保留注释写回 → `scheduler.maybe_reload`」。REST API 返回 JSON；页面路由返回服务端渲染的 HTML（Jinja2），HTMX 做局部刷新。可选 `WEB_PASSWORD` 环境变量 → HTTP Basic Auth。`serve` 命令重写成 `uvicorn.run(app)`。

**Tech Stack:** FastAPI、uvicorn[standard]、Jinja2、python-multipart（表单）、HTMX（vendored 静态文件）。测试用 `fastapi.testclient.TestClient` + `respx`。

**Spec:** `docs/superpowers/specs/2026-08-31-push2xteink-design.md` 第 8 节（Web 界面 + REST API 列表 + 配置写回）、第 3/10 节（单进程入口）、第 13 节（cron 星期语义 → 人类可读预览）、第 14 节全部「P5 需要...」条目。

## Global Constraints

- Python `>=3.12`；`X | None` / `list[...]` 标注。
- **所有 HTTP 出站请求走 `push2xteink.http.make_client`**（spec §14）——测试连接的 AI/xteink/proxy 探测也是。
- 配置写回**只经 `config.py` 的 `write_config` / `load_config`**——不要在 web 层直接碰 yaml。每次写回后调 `scheduler.maybe_reload(config_path)`。
- 任何配置改动，应用后必须整体 `Config.model_validate`（或构造 `Config(...)`）重校验；`ValidationError` → HTTP 400 附可读消息，**不写文件**。
- `Scheduler` / `State` 由 lifespan 拥有；路由通过 `request.app.state.scheduler` / `.db_state` / `.config_path` 取。
- 敏感字段（`xteink.password`、`ai.primary.api_key`、`ai.fallback.api_key`）在 `GET /api/settings` 返回时替换成 `"********"`；`PUT` 时如果收到的值等于 `"********"` 则保留原值。
- 手动触发用 `Scheduler.submit(task_id)`（异步、走 APS executor），**不用** `run_now`（同步阻塞请求）。
- 不引入前端构建链。HTMX 作为 vendored 静态文件（`web/static/htmx.min.js`）。CSS 内联在 base 模板里，够用即可。
- 源文件 `src/push2xteink/web/`；模板 `src/push2xteink/web/templates/`；静态 `src/push2xteink/web/static/`。测试 `tests/`。

_PROXY_SCHEMES = ("http://", "https://", "socks5://", "socks5h://", "socks4://")

class ProxyConfig(BaseModel):

```python
from datetime import datetime
from apscheduler.triggers.date import DateTrigger
from .config import ConfigError, load_config

# __init__: self._config_token: tuple[int, int] | None = None
#           self._last_bad_token: tuple[int, int] | None = None

def submit(self, task_id: str) -> None:
    self._aps.add_job(
        self._run, DateTrigger(run_date=datetime.now(self._aps.timezone)),
        args=[task_id], id=f"manual:{task_id}", replace_existing=True,
        misfire_grace_time=60,
    )

def _token(self, path) -> tuple[int, int] | None:
    try:
        st = Path(path).stat()
        return (st.st_mtime_ns, st.st_size)
    except OSError:
        return None

def maybe_reload(self, config_path) -> bool:
    token = self._token(config_path)
    if token is None or token == self._config_token:
        return False
    try:
        new_config = load_config(Path(config_path))
    except ConfigError as exc:
        if token != self._last_bad_token:
            logger.warning("config reload skipped (invalid): %s", exc)
            self._last_bad_token = token
        return False
    self.reload(new_config)
    self._config_token = token
    return True
```

（`__init__` 里初始化时若已有 config 文件，可不预置 token——首次 `maybe_reload` 会 reload 一次，等价于无操作重建；或在 `serve`/lifespan 里传入初始 path 让 `Scheduler` 预置 token。简单起见：`Scheduler.__init__` 不管 token；lifespan 在 `start()` 后立刻 `sched._config_token = sched._token(config_path)` —— 或加一个 `Scheduler.prime_config_token(path)`。**实现 `prime_config_token(path)`**，lifespan 和 serve 调它。）

```python
from __future__ import annotations
import logging, threading
from pathlib import Path
from typing import Callable

logger = logging.getLogger(__name__)


class ConfigWatcher:
    def __init__(self, path: Path, on_tick: Callable[[], object], *, interval: float = 5.0) -> None:
        self._path = Path(path)
        self._on_tick = on_tick
        self._interval = interval
        self._stop = threading.Event()
        self._thread: threading.Thread | None = None

    def start(self) -> None:
        if self._thread is not None:
            return
        self._stop.clear()
        self._thread = threading.Thread(target=self._loop, name="config-watcher", daemon=True)
        self._thread.start()

    def _loop(self) -> None:
        while not self._stop.wait(self._interval):
            try:
                self._on_tick()
            except Exception:
                logger.exception("config watcher on_tick failed")

    def stop(self) -> None:
        self._stop.set()
        if self._thread is not None:
            self._thread.join(timeout=self._interval + 5)
            self._thread = None
```

def _auth_dep(creds: HTTPBasicCredentials | None = Depends(_basic)) -> None:
    pw = os.environ.get("WEB_PASSWORD")
    if not pw:
        return
    if creds is None or not secrets.compare_digest(creds.password, pw):

def create_app(config_path: Path, db_path: Path) -> FastAPI:
    config_path, db_path = Path(config_path), Path(db_path)

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        db_path.parent.mkdir(parents=True, exist_ok=True)
        state = State(db_path)
        sched = Scheduler(load_config(config_path), state)
        sched.start()
        sched.prime_config_token(config_path)
        watcher = ConfigWatcher(config_path, lambda: sched.maybe_reload(config_path))
        watcher.start()
        app.state.db_state = state
        app.state.scheduler = sched
        app.state.config_path = config_path
        try:
            yield
        finally:
            watcher.stop()
            sched.shutdown()
            state.close()

    app = FastAPI(lifespan=lifespan, dependencies=[Depends(_auth_dep)])
    app.mount("/static", StaticFiles(directory=str(_HERE / "static")), name="static")

    @app.get("/healthz")

    from .api_settings import router as settings_router
    from .api_runs import router as runs_router
    from .pages import router as pages_router

from apscheduler.triggers.cron import CronTrigger
def _preview(expr: str) -> dict:
    try:
        trig = CronTrigger.from_crontab(expr)
    except ValueError as e:
        return {"valid": False, "next": [], "error": str(e)}
    out, prev = [], None
    for _ in range(3):
        nxt = trig.get_next_fire_time(prev, datetime.now(timezone.utc))
        if nxt is None: break
        out.append(nxt.isoformat()); prev = nxt
    return {"valid": True, "next": out, "error": None}
